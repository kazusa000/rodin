cmake_minimum_required(VERSION 3.16)
project(KelvinOriginalCoreGraded LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(RODIN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(RODIN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(RODIN_BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(RODIN_BUILD_MANUFACTURED_TESTS OFF CACHE BOOL "" FORCE)
set(RODIN_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(RODIN_BUILD_DOC OFF CACHE BOOL "" FORCE)
set(RODIN_INSTALL_RESOURCES OFF CACHE BOOL "" FORCE)
set(RODIN_WITH_PY OFF CACHE BOOL "" FORCE)
set(RODIN_USE_PETSC OFF CACHE BOOL "" FORCE)
set(RODIN_USE_MPI OFF CACHE BOOL "" FORCE)
set(RODIN_USE_MUMPS ON CACHE BOOL "" FORCE)
set(RODIN_MULTITHREADED ON CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${RODIN_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/rodin")
set(CARLOS_MUMPS_SUPPORT_LIBRARIES "" CACHE STRING "Existing MUMPS link dependencies")
if(CARLOS_MUMPS_SUPPORT_LIBRARIES)
  target_link_libraries(RodinSolver INTERFACE ${CARLOS_MUMPS_SUPPORT_LIBRARIES})
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}/kelvin")
set_target_properties(KelvinBall KelvinBallSphere PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
add_executable(KelvinBallGradedDomainTest ../GradedDomainTest.cpp ../Configuration.cpp)
target_include_directories(KelvinBallGradedDomainTest PRIVATE "${CMAKE_CURRENT_LIST_DIR}/..")
target_link_libraries(KelvinBallGradedDomainTest PRIVATE Rodin::Geometry)
set_target_properties(KelvinBallGradedDomainTest PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
enable_testing()
add_test(NAME KelvinBallGradedDomainTest COMMAND KelvinBallGradedDomainTest)
