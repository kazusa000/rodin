cmake_minimum_required(VERSION 3.16)
project(KelvinFemOld LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(RODIN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
foreach(option IN ITEMS RODIN_BUILD_EXAMPLES RODIN_BUILD_UNIT_TESTS
    RODIN_BUILD_MANUFACTURED_TESTS RODIN_BUILD_BENCHMARKS RODIN_BUILD_DOC
    RODIN_INSTALL_RESOURCES RODIN_WITH_PY RODIN_USE_PETSC RODIN_USE_MPI BUILD_TESTING)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(RODIN_USE_MUMPS ON CACHE BOOL "" FORCE)
set(RODIN_MULTITHREADED ON CACHE BOOL "" FORCE)
add_subdirectory("${RODIN_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/rodin")
set(CARLOS_MUMPS_SUPPORT_LIBRARIES "" CACHE STRING "Existing MUMPS link dependencies")
if(CARLOS_MUMPS_SUPPORT_LIBRARIES)
  target_link_libraries(RodinSolver INTERFACE ${CARLOS_MUMPS_SUPPORT_LIBRARIES})
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/.." "${CMAKE_CURRENT_BINARY_DIR}/kelvin")
set_target_properties(KelvinBall KelvinBallCubicGeometryTest KelvinBallTetraGeometryTest
  PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
